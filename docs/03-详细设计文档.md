## 一、项目概述

### 1.1 项目信息

| 项目 | 内容 |
|------|------|
| 项目名称 | QNote |
| 项目类型 | 桌面便签应用 |
| 版本号 | v1.0 |
| 技术栈 | Qt 6.5+ / C++17 / SQLite 3 / CMake 3.16+ / md4c |
| 目标平台 | Linux (Kylin/UOS/Ubuntu), Windows 10/11 |
| 许可证 | MIT / LGPL |

### 1.2 项目目标

- **轻量快速**：启动时间 < 1秒，内存占用 < 50MB
- **Markdown 原生**：支持完整的 Markdown 编辑与实时预览
- **本地优先**：数据存储在本地，保护隐私
- **跨平台**：一套代码，多平台运行
- **国产适配**：优先支持 Kylin/UOS 国产操作系统

```sql
CREATE TABLE notes (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    uuid            TEXT UNIQUE NOT NULL,
    title           TEXT DEFAULT '',
    content         TEXT DEFAULT '',
    content_type    TEXT DEFAULT 'markdown' CHECK(content_type IN ('plain', 'markdown', 'html')),
    color           TEXT DEFAULT '#FFFFFF',
    is_pinned       INTEGER DEFAULT 0 CHECK(is_pinned IN (0, 1)),
    is_deleted      INTEGER DEFAULT 0 CHECK(is_deleted IN (0, 1)),
    is_archived     INTEGER DEFAULT 0 CHECK(is_archived IN (0, 1)),
    group_id        INTEGER REFERENCES groups(id) ON DELETE SET NULL,
    word_count      INTEGER DEFAULT 0,
    char_count      INTEGER DEFAULT 0,
    created_at      INTEGER NOT NULL,
    updated_at      INTEGER NOT NULL,
    deleted_at      INTEGER,
    version         INTEGER DEFAULT 1,
    sync_status     TEXT DEFAULT 'synced' CHECK(sync_status IN ('synced', 'pending', 'conflict'))
);

-- 索引
CREATE INDEX idx_notes_created ON notes(created_at DESC);
CREATE INDEX idx_notes_updated ON notes(updated_at DESC);
CREATE INDEX idx_notes_pinned ON notes(is_pinned, updated_at DESC);
CREATE INDEX idx_notes_deleted ON notes(is_deleted);
CREATE INDEX idx_notes_group ON notes(group_id);
CREATE INDEX idx_notes_uuid ON notes(uuid);
CREATE INDEX idx_notes_sync ON notes(sync_status);

-- 全文搜索虚拟表（FTS5）
CREATE VIRTUAL TABLE notes_fts USING fts5(
    title,
    content,
    content='notes',
    content_rowid='id',
    tokenize='unicode61'
);

-- 触发器：自动更新 FTS 索引
CREATE TRIGGER notes_ai AFTER INSERT ON notes BEGIN
    INSERT INTO notes_fts(rowid, title, content) VALUES (new.id, new.title, new.content);
END;

CREATE TRIGGER notes_ad AFTER DELETE ON notes BEGIN
    INSERT INTO notes_fts(notes_fts, rowid, title, content) VALUES('delete', old.id, old.title, old.content);
END;

CREATE TRIGGER notes_au AFTER UPDATE ON notes BEGIN
    INSERT INTO notes_fts(notes_fts, rowid, title, content) VALUES('delete', old.id, old.title, old.content);
    INSERT INTO notes_fts(rowid, title, content) VALUES (new.id, new.title, new.content);
END;
```

### 3.3 视图设计

```sql
-- 便签列表视图（包含标签和分组信息）
CREATE VIEW v_notes_list AS
SELECT 
    n.id, n.uuid, n.title, n.content, n.content_type, n.color,
    n.is_pinned, n.is_deleted, n.is_archived, n.group_id,
    n.word_count, n.char_count, n.created_at, n.updated_at,
    g.name as group_name, g.color as group_color,
    GROUP_CONCAT(t.name, ',') as tag_names,
    GROUP_CONCAT(t.color, ',') as tag_colors
FROM notes n
LEFT JOIN groups g ON n.group_id = g.id
LEFT JOIN note_tags nt ON n.id = nt.note_id
LEFT JOIN tags t ON nt.tag_id = t.id
WHERE n.is_deleted = 0
GROUP BY n.id;

-- 待处理提醒视图
CREATE VIEW v_pending_reminders AS
SELECT 
    r.id, r.note_id, r.remind_at, r.repeat_type, r.snooze_count,
    n.title, n.content
FROM reminders r
JOIN notes n ON r.note_id = n.id
WHERE r.is_done = 0 AND r.remind_at <= strftime('%s', 'now') + 300
ORDER BY r.remind_at ASC;
```

### 3.4 触发器

```sql
-- 更新便签时自动更新 updated_at
CREATE TRIGGER notes_update_time 
AFTER UPDATE ON notes
BEGIN
    UPDATE notes SET updated_at = strftime('%s', 'now') WHERE id = NEW.id;
END;

-- 更新标签的便签计数
CREATE TRIGGER note_tags_count_inc
AFTER INSERT ON note_tags
BEGIN
    UPDATE tags SET note_count = note_count + 1 WHERE id = NEW.tag_id;
END;

CREATE TRIGGER note_tags_count_dec
AFTER DELETE ON note_tags
BEGIN
    UPDATE tags SET note_count = note_count - 1 WHERE id = OLD.tag_id;
END;

-- 删除便签时清理相关数据
CREATE TRIGGER notes_cleanup
AFTER UPDATE OF is_deleted ON notes
WHEN NEW.is_deleted = 1
BEGIN
    UPDATE notes SET deleted_at = strftime('%s', 'now') WHERE id = NEW.id;
    DELETE FROM note_tags WHERE note_id = NEW.id;
    DELETE FROM reminders WHERE note_id = NEW.id;
END;
```
